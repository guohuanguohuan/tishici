\xszhang{第九章\quad 静电场及其应用}
\xsguanlian{9.1 电荷；9.2 库仑定律；9.3 电场\quad 电场强度；9.4 静电的防止与利用}
\xszhuxian{从「两类电」的辨认、正负电荷的命名，到库仑用扭秤定量测力，再到密立根测出最小电荷单位，电学由定性观察走向精密测量。}
\xslabel{【关键公式】}
{\noindent\hangindent=8.3mm\hangafter=1{\kaishu \huidi{库仑定律}$F=\dfrac{k|q_1 q_2|}{r^{2}}$（条件：\huidi{真空中的点电荷}；方向沿两电荷连线）；\huidi{元电荷}$e=1.602\,176\,634\times10^{-19}\,\mathrm{C}$（教材常用近似值 $1.60\times10^{-19}\,\mathrm{C}$）．}\par}
\tiaomu{1}{\biaoqian{【必记】}\huidi{18世纪30年代}，\huidi{杜菲}区分「两类电」，并总结\huidi{同类相斥、异类相吸}；\huidi{富兰克林}后来采用\huidi{正、负电荷的命名}并研究雷电（杜菲条系史实补充，教材未提及；必修三9.1节）。}
\tiaomu{2}{\biaoqian{【必记】}\huidi{富兰克林}通过电学研究，论证\huidi{闪电是一种放电现象}，并推动\huidi{避雷针}的应用（必修三9.1节、9.4节）。}
\tiaomu{3}{\biaoqian{【必记】}\huidi{1785年}，法国科学家\huidi{库仑}设计了精妙的\huidi{扭秤实验}研究电荷间的作用力，建立\huidi{库仑定律}（必修三9.2节）。}
\tiaomu{4}{\biaoqian{【必记】}元电荷$e$的数值最早是由美国物理学家\huidi{密立根}通过（油滴）实验\huidi{测得}的，其关键结果约在\huidi{1910年前后}发表，并获\huidi{诺贝尔物理学奖}（1923年颁奖，为史实补充；必修三9.1节）。}
\tiaomu{5}{\biaoqian{【易错】}不写「富兰克林发现自然界有两种电荷」——教材口径是富兰克林\huidi{命名正负电荷}；静电力常量$k$的测定不宜归给库仑或高斯，高考以\huidi{库仑扭秤实验和库仑定律}为稳妥口径。}
