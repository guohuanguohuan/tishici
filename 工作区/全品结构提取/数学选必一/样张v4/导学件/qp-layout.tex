\usepackage{amsmath,amssymb}
\usepackage{graphicx}
\usepackage{xcolor}
\usepackage{colortbl}

\setlength{\columnseprule}{0.4pt}      % 栏间竖线（全品 p04-p06 实证宽约 0.17mm 近似 0.4pt）

\normalsize
\setlength{\parindent}{0pt}             % 顶格化（v4.1 指令5）：全品正文无首行缩进

\widowpenalty=10000
