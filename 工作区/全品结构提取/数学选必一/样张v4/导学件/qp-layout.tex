% qp-layout.tex —— 版面模块（全品式导学件，供 main.tex \input）
% 职责：几何（A4 左/右/顶 15mm、底 20mm（v4.2 拍板22 让位页脚带）、twoside 对称；v4.2 headheight/headsep=0pt 撤页眉位）、宏包、
%       调色板（黑＋777777＋DDDDDD＋black!25 栏线派生档拍板39）、
%       行距/栏距/栏线/断行惩罚/浮动参数、表格默认、顶格化口径（\parindent=0pt）。
% 调用关系：main.tex → qp-fonts → 本模块 → qp-headfoot → qp-titles → qp-blocks。
% ============================================================
% v4.2 拍板22 冲突实害修复：bottom=20mm 正文区让位页脚带（全品 p04/p05 实测正文最低≈22.8mm、
% 块底≈11.4mm——全品正文区让开页脚带；v4.2 原 margin=15mm 实证 p1 页脚盖正文末行，故底边让 5mm）
\usepackage[a4paper,margin=15mm,bottom=20mm,twoside,headheight=0pt,headsep=0pt,footskip=13pt]{geometry}
\usepackage{amsmath,amssymb}
\usepackage{graphicx}
\usepackage{xcolor}
\usepackage{multirow,array,calc}
\usepackage{multicol}

% ---------- 颜色（调色板收敛：黑＋777777＋DDDDDD，单页灰档≤2，拍板27） ----------
\definecolor{midgray}{HTML}{777777}    % 灰字/括注/注记/灰小字/章首方块/花形底线（唯一中灰）
\definecolor{pnumbg}{HTML}{DDDDDD}     % 页码灰块/章首浅方块（唯一浅灰）

% ---------- 版面 ----------
\setlength{\columnsep}{7.5mm}
\setlength{\columnseprule}{0.4pt}      % 栏间竖线恢复（v4.2 翻拍板28：全品 p04-p06 实证页心竖线，宽约 0.17mm 近似 0.4pt）
\renewcommand{\columnseprulecolor}{\color{black!25}}   % 全品实测灰度 188≈black!25（0xBF，拍板39 调色板派生档）
\setlength{\multicolsep}{0pt}         % 章首【学习目标】标签→栏区零距（④ 36.2→34.4mm 达 ≤35 门；multicol 前后 skip 全局归零，单 multicols 无副作用）
\renewcommand{\normalsize}{\fontsize{10.5pt}{18pt}\selectfont}
\linespread{1}                          % 行距通解：撤 ctex linestretch 1.3 叠乘，实发 18pt
\normalsize
\setlength{\parindent}{0pt}             % 顶格化（v4.1 指令5）：全品正文无首行缩进
\clubpenalty=10000
\widowpenalty=10000
\displaywidowpenalty=10000
\tolerance=9999
\emergencystretch=8em   % 序言预设；实际生效处在 multicols 内（multicol 进入时会重置为 8pt）
\binoppenalty=100
\relpenalty=100
\flushbottom            % 对照§四-5（\raggedbottom 撤）
\raggedcolumns          % multicol 栏末不硬拉伸（末页参差收尾，拍板23）
\setcounter{topnumber}{4}
\setcounter{totalnumber}{4}
\setcounter{bottomnumber}{2}
\renewcommand{\topfraction}{0.95}
\renewcommand{\textfraction}{0.05}
\renewcommand{\floatpagefraction}{0.85}

% ---------- 表格默认（全框细线＋紧凑内边距；arraystretch 1.05，对照§四-6；v4.2 1.15→1.05：bottom=20mm
%           后 p1 左栏剩余约 81mm 装不下 1.15 档表1（86.2mm），1.05 省约 5.8mm 使表1 归位左栏） ----------
\setlength{\arrayrulewidth}{0.4pt}
\setlength{\tabcolsep}{3pt}
\renewcommand{\arraystretch}{1.05}
\renewcommand{\multirowsetup}{\centering}
