%       SimHei 无真粗体面，以 FakeBold=2.2 合成（实测墨密 0.292→0.391，仅四级标题用；
%       正文内【】标签、◆标题维持 \heiti/\bfseries 口径不动）。

\setCJKmainfont[BoldFont=SimHei]{SimSun}

\setCJKfamilyfont{heibf}[FakeBold=2.2]{SimHei}
\setCJKfamilyfont{kai}{KaiTi}

\newcommand{\heibf}{\CJKfamily{heibf}}
\newcommand{\kaishu}{\CJKfamily{kai}}
