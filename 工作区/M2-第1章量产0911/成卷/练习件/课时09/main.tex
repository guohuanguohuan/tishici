\dangbadge{基}{础}{巩}{固}

\tihao{1}\zu{J1 射影面积法二面角×直算填空〔代表〕} \tieside{简单}已知二面角\(P\text{-}\penalty100 AB\text{-}\penalty100 P'\)的大小为\(\frac{\pi}{6}\)，且\(PP'\perp\)平面\(ABP'\)，\(\triangle ABP\)的面积为\(3\)，求\(\triangle ABP'\)的面积．
\xiexwei{16mm}

\tihao{2}\zu{J2 直二面角·勾股链\(CD^{2}\)源头×填空} \tieside{简单}已知直二面角\(\alpha\text{-}\penalty100 l\text{-}\penalty100 \beta\)的棱上有\(A\)，\(B\)两个点，\(AC\subset\alpha\)，\(AC\perp l\)，\(BD\subset\beta\)，\(BD\perp l\)，若\(AB=5\)，\(AC=3\)，\(BD=8\)，求\(CD\)的长．
\xiexwei{16mm}

\tihao{3}\zu{J3 建系向量法·先垂直后建系×双问〔代表〕（取锐）} \tieside{中档}如图所示的几何体中，\(BE\perp BC\)，\(EA\perp AC\)，\(BC=2\)，\(AC=2\sqrt{2}\)，\(\angle ACB=45^\circ\)，\(AD\parallel BC\)，\(BC=2AD\)．
\par\vspace{1.9mm}\penalty10000\noindent\makebox[\linewidth][c]{\includegraphics[width=30mm]{figs/js_image88.png}}\par\vspace{-1.0mm}\penalty10000
\lxopt{(1)求证：\(AE\perp\)平面\(ABCD\)；}
\lxopt{(2)若\(\angle ABE=60^\circ\)，点\(F\)在\(EC\)上，且满足\(EF=2FC\)，求平面\(FAD\)与平面\(ADC\)的夹角的余弦值．}
\xiexwei{20mm}

\tihao{4}\zu{J3〔第2题〕锐钝判定「同负异补」×双问（取钝）} \tieside{中档偏上}在四棱锥\(P\text{-}\penalty100 ABCD\)中，\(AB\parallel CD\)，且\(\angle BAP=\angle CDP=90^\circ\)．
\lxopt{(1)证明：平面\(PAB\perp\)面\(PAD\)；}
\lxopt{(2)若\(PA=PD=AB=DC\)，\(\angle APD=90^\circ\)，求二面角\(A\text{-}\penalty100 PB\text{-}\penalty100 C\)的余弦值．}
\xiexwei{20mm}

\tihao{5}\zu{J4 圆锥载体·四法三法方法总览×双问} \tieside{中档}如图，\(D\)为圆锥的顶点，\(O\)是圆锥底面的圆心，\(AE\)为底面直径，\(AE=AD\)．\(\triangle ABC\)是底面的内接正三角形，\(P\)为\(DO\)上一点，\(PO=\frac{\sqrt{6}}{6}DO\)．
\par\vspace{1.9mm}\penalty10000\noindent\makebox[\linewidth][c]{\includegraphics[width=25mm]{figs/js_image103.png}}\par\vspace{-1.0mm}\penalty10000
\lxopt{(1)证明：\(PA\perp\)平面\(PBC\)；}
\lxopt{(2)求二面角\(B\text{-}\penalty100 PC\text{-}\penalty100 E\)的余弦值．}
\xiexwei{24mm}

\tihao{6}\zu{J5 二面角含参存在性×菱形底两问〔代表〕} \tieside{中档}如图，在四棱锥\(P\text{-}\penalty100 ABCD\)中，底面\(ABCD\)是边长为\(2\)的菱形，\(\angle ABC=60^\circ\)，\(\triangle PAB\)为正三角形，且侧面\(PAB\perp\)底面\(ABCD\)，\(E\)为线段\(AB\)的中点，\(M\)在线段\(PD\)上．
\par\vspace{1.9mm}\penalty10000\noindent\makebox[\linewidth][c]{\includegraphics[width=32mm]{figs/dy6_image47.png}}\par\vspace{-1.0mm}\penalty10000
\lxopt{(1)求证：\(PE\perp AC\)；}
\lxopt{(2)是否存在点\(M\)，使二面角\(M\text{-}\penalty100 EC\text{-}\penalty100 D\)的大小为\(60^\circ\)，若存在，求出\(\frac{PM}{PD}\)的值；若不存在，请说明理由．}
\xiexwei{26mm}

\tihao{7}\zu{J6 距离＋二面角综合×菱形底大题〔代表·C组难槽〕} \tieside{难}如图所示，已知四棱锥\(P\text{-}\penalty100 ABCD\)中，\(PB\perp AD\)，侧面\(PAD\)为边长等于\(2\)的正三角形，底面\(ABCD\)为菱形，侧面\(PAD\)与底面\(ABCD\)所成的二面角为\(120^\circ\)．
\lxopt{(1)求点\(P\)到平面\(ABCD\)的距离；}
\lxopt{(2)求二面角\(A\text{-}\penalty100 PB\text{-}\penalty100 C\)的余弦值．}
\xiexwei{40mm}

\tihao{8}\zu{J7 垂直＋二面角＋存在性三问×正四棱锥〔C组难槽·自1.2.2移交〕} \tieside{难}如图所示，四棱锥\(S\text{-}\penalty100 ABCD\)的底面是正方形，每条侧棱的长都是底面边长的\(\sqrt{2}\)倍，\(P\)为侧棱\(SD\)上的点．
\lxopt{(1)求证：\(AC\perp SD\)；}
\lxopt{(2)若\(SD\perp\)平面\(PAC\)，求二面角\(P\text{-}\penalty100 AC\text{-}\penalty100 D\)的大小；}
\lxopt{(3)在(2)的条件下，侧棱\(SC\)上是否存在一点\(E\)，使得\(BE\parallel\)平面\(PAC\)．若存在，求\(SE:EC\)的值；若不存在，试说明理由．}
\xiexwei{48mm}

\tihao{9}\zu{二面角平面角·定义辨认×选择〔代表〕} \tieside{简单}已知\(P\)是二面角\(\alpha\text{-}\penalty100 l\text{-}\penalty100 \beta\)的棱\(l\)上一点，\(M\)是半平面\(\alpha\)内一点，\(N\)是半平面\(\beta\)内一点（\(M\)，\(N\)均与\(P\)不重合）．则下列条件中，能使\(\angle MPN\)成为二面角\(\alpha\text{-}\penalty100 l\text{-}\penalty100 \beta\)的平面角的是\nobreak(\kongwei)
\lxopt{\makebox[\dimexpr0.5\linewidth-0.5\lxhang-0.25em\relax][l]{A．\(MP\perp l\)；}\makebox[\dimexpr0.5\linewidth-0.5\lxhang-0.25em\relax][l]{B．\(NP\perp l\)；}}
\lxopt{\makebox[\dimexpr0.5\linewidth-0.5\lxhang-0.25em\relax][l]{C．\(MP\perp l\)且\(NP\perp l\)；}\makebox[\dimexpr0.5\linewidth-0.5\lxhang-0.25em\relax][l]{D．\(MP\perp NP\)}}

\tihao{10}\zu{勾股链·平面角直读×填空〔代表〕} \tieside{简单}在二面角\(\alpha\text{-}\penalty100 l\text{-}\penalty100 \beta\)中，棱\(l\)上取一点\(A\)，\(AB\subset\alpha\)且\(AB\perp l\)于点\(A\)，\(AC\subset\beta\)且\(AC\perp l\)于点\(A\)．若\(AB=AC=1\)，\(BC=\sqrt{2}\)，则二面角\(\alpha\text{-}\penalty100 l\text{-}\penalty100 \beta\)的大小为\kongbai{}．

\dangbadge{综}{合}{提}{升}

\tihao{11}\zu{直二面角·性质垂直链×填空〔代表〕} \tieside{简单}在直二面角\(\alpha\text{-}\penalty100 l\text{-}\penalty100 \beta\)中，点\(B\)在棱\(l\)上，\(BA\subset\alpha\)且\(BA\perp l\)于点\(B\)，\(BA=3\)．点\(D\)在半平面\(\beta\)内，\(BD=4\)，且\(BD\)与棱\(l\)所成的角为\(60^\circ\)．则\(AD=\)\kongbai{}．

\tihao{12}\zu{射影面积公式一步直取×选择〔代表〕} \tieside{简单}已知一个边长为\(2\)的正三角形位于二面角\(\alpha\text{-}\penalty100 l\text{-}\penalty100 \beta\)的半平面\(\alpha\)内，它在半平面\(\beta\)内的射影图形的面积为\(\frac{\sqrt{3}}{2}\)．则二面角\(\alpha\text{-}\penalty100 l\text{-}\penalty100 \beta\)的大小为\nobreak(\kongwei)
\lxopt{\makebox[\dimexpr0.25\linewidth-0.25\lxhang-0.25em\relax][l]{A．\(30^\circ\)；}\makebox[\dimexpr0.25\linewidth-0.25\lxhang-0.25em\relax][l]{B．\(45^\circ\)；}\makebox[\dimexpr0.25\linewidth-0.25\lxhang-0.25em\relax][l]{C．\(60^\circ\)；}\makebox[\dimexpr0.25\linewidth-0.25\lxhang-0.25em\relax][l]{D．\(90^\circ\)}}

\tihao{13}\zu{射影面积公式一步直取〔第2题〕×填空} \tieside{简单}二面角\(\alpha\text{-}\penalty100 l\text{-}\penalty100 \beta\)的大小为\(60^\circ\)，半平面\(\alpha\)内一个图形的面积记为\(S\)．若\(S\)在半平面\(\beta\)内的射影图形恰为两条直角边长分别为\(3\)和\(4\)的直角三角形，则\(S=\)\kongbai{}．

\tihao{14}\zu{勾股链·平面角直读〔第2题〕×填空} \tieside{简单}在三棱锥\(P\text{-}\penalty100 ABC\)中，\(PA\perp\)平面\(ABC\)，\(\triangle ABC\)是边长为\(2\)的正三角形，\(D\)为\(BC\)的中点，连接\(AD\)，\(PD\)．若\(PA=3\)，则二面角\(P\text{-}\penalty100 BC\text{-}\penalty100 A\)的大小为\kongbai{}．

\dangbadge{思}{维}{探}{索}

\tihao{15}\zu{直二面角·性质垂直链〔第2题〕×解答} \tieside{简单}已知直二面角\(\alpha\text{-}\penalty100 l\text{-}\penalty100 \beta\)，点\(A\)在棱\(l\)上，点\(B\)在半平面\(\alpha\)内，点\(C\)在半平面\(\beta\)内，且\(BA\perp l\)于点\(A\)，\(CA\perp l\)于点\(A\)．
\lxopt{(1)求证：\(AB\perp\)平面\(\beta\)；}
\lxopt{(2)若\(AB=1\)，\(BC=2\)，求\(\angle ACB\)的大小．}
\xiexwei{70mm}

\tihao{16}\zu{二面角平面角·定义辨认〔第2题〕×选择} \tieside{简单}在正方体\(ABCD\text{-}\penalty100 A_1B_1C_1D_1\)中，二面角\(C_1\text{-}\penalty100 AB\text{-}\penalty100 C\)的大小为\nobreak(\kongwei)
\lxopt{\makebox[\dimexpr0.25\linewidth-0.25\lxhang-0.25em\relax][l]{A．\(30^\circ\)；}\makebox[\dimexpr0.25\linewidth-0.25\lxhang-0.25em\relax][l]{B．\(45^\circ\)；}\makebox[\dimexpr0.25\linewidth-0.25\lxhang-0.25em\relax][l]{C．\(60^\circ\)；}\makebox[\dimexpr0.25\linewidth-0.25\lxhang-0.25em\relax][l]{D．\(90^\circ\)}}
