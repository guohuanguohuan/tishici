\emergencystretch=1em

% ===== 课前预习（逐空对号） =====
\qufen{课前预习}{第1课时　序与导学件同}

\kdhead{知识点一　电动势}{填空7空／判断2题}
\ansline{答案}{①$\frac{W}{q}$　②标　③其他形式的能　④2　⑤电源　⑥电场　⑦电压；判断：(1)√　(2)×}
\ansline{解析}{判断(1)：电动势由电源自身的性质决定，与外电路的组成无关；判断(2)：电压表测得的是电源的路端电压，因内阻上有分压，其示数总略小于电动势，只有断路时路端电压才等于电动势．}

\kdhead{知识点二　闭合电路的欧姆定律}{填空4空／判断2题}
\ansline{答案}{①$U+U_{\text{内}}$　②$\frac{E}{R+r}$　③$E-Ir$　④增大；判断：(1)×　(2)×}
\ansline{解析}{判断(1)：断路时 $I=0$，内电压为零，路端电压 $U=E$；判断(2)：外电路含非纯电阻元件时，欧姆定律对整个闭合电路不适用，应按串、并联电路特点或能量守恒定律列式．}

\kdhead{知识点三　电源的$U$-$I$图像}{填空3空／判断1题}
\ansline{答案}{①$E$　②短路电流$\frac{E}{r}$　③$r$；判断：(1)×}
\ansline{解析}{图线斜率的绝对值等于电源的内阻 $r$，斜率的绝对值越大，内阻越大．}

% ===== 课中探究 =====
\qufen{课中探究}{探究点一　例1、变式1；探究点二　例2—例3、变式2}

\anslx{例1}{A}
\ansline{知识点}{电源、电动势的定义、电动势与电势差的对比、闭合电路欧姆定律的内容及公式}
\ansline{详解}{A．电源电动势反映电源将其他形式能量转化为电能的本领大小，电源电动势越大说明电源把其他形式的能转化为电能的本领越大，A正确；\par
B．用电压表直接测量电源两极得到的电压数值，实际上是电源的路端电压，总略小于电源电动势的准确值，B错误；\par
C．电源电动势总等于内、外电路上的电压之和，但它的数值与外电路的组成无关，电动势反映电源的特性，C错误；\par
D．电源正、负极之间的电势差为电源的路端电压，小于电源电动势，只有当电源处于断路状态时，电源的电动势才等于路端电压，D错误．\par
故选A．}

\anslx{变式1}{D}
\ansline{解析}{A．$E=\frac{W}{q}$为定义式而非决定式，$E$ 由电源自身的性质决定，与 $W$、$q$ 无关，A错误；\par
B．正、负极短接不改变电源自身的性质，电动势不变，B错误；\par
C．电动势是标量（电源内部电流方向由负极指向正极），C错误；\par
D．1号干电池电动势约 $1.5\,\unit{V}$，小于铅蓄电池的 $2\,\unit{V}$，表明前者把化学能转化为电能的本领较弱，D正确．\par
故选D．}

\anslx{例2}{(1)$2\,\unit{A}$；(2)$1\,\unit{\upOmega}$}
\ansline{知识点}{闭合电路欧姆定律的内容及公式、电功和电功率定义、表达式及简单应用}
\ansline{详解}{(1) 两灯恰能正常发光，每盏灯两端电压 $U_L=6\,\unit{V}$，通过每盏灯的电流\par
$I_1=\frac{P_L}{U_L}=\frac{12}{6}\,\unit{A}=2\,\unit{A}$．\par
(2) 两灯并联，干路电流 $I=2I_1=4\,\unit{A}$．由闭合电路的欧姆定律\par
$E=U_L+Ir$，得 $r=\frac{E-U_L}{I}=\frac{10-6}{4}\,\unit{\upOmega}=1\,\unit{\upOmega}$．}

\anslx{拓展延伸}{不能正常发光}
\ansline{详解}{每盏灯的电阻 $R_L=\frac{U^2}{P}=\frac{6^2}{12}\,\unit{\upOmega}=3\,\unit{\upOmega}$；串联后外电路总电阻 $R_{\text{外}}=2R_L=6\,\unit{\upOmega}$．\par
电路电流 $I=\frac{E}{R_{\text{外}}+r}=\frac{10}{6+1}\,\unit{A}\approx1.4\,\unit{A}$，每盏灯两端电压 $U'=IR_L\approx4.3\,\unit{V}<6\,\unit{V}$（实际功率 $P'=I^2R_L\approx6.1\,\unit{W}<12\,\unit{W}$），故两灯不能正常发光．}

\anslx{例3}{B}
\ansline{知识点}{电源的$U$-$I$图像}
\ansline{详解}{A．两图线的交点即电阻 $R$ 接在该电源上的工作点，由图可知电源的输出电压为 $1.5\,\unit{V}$，A错误；\par
B．由图线Ⅰ可知，电源的电动势为 $3.0\,\unit{V}$，内阻 $r=\left|\frac{\Delta U}{\Delta I}\right|=\frac{3.0}{6.0}\,\unit{\upOmega}=0.5\,\unit{\upOmega}$，B正确；\par
C．电源的输出功率 $P=UI=3.0\times1.5\,\unit{W}=4.5\,\unit{W}$，C错误；\par
D．电源的效率 $\eta=\frac{U}{E}\times100\%=50\%$，D错误．\par
故选B．}

\anslx{变式2}{$E=4.5\,\unit{V}$；$r=1.5\,\unit{\upOmega}$}
\ansline{解析}{图线与纵轴的截距等于电源的电动势，$E=4.5\,\unit{V}$；图线斜率的绝对值等于内阻，$r=\frac{4.5}{3.0}\,\unit{\upOmega}=1.5\,\unit{\upOmega}$．}

% ===== 课堂评价 =====
\qufen{课堂评价}{本组共5题（时间:15分钟　总分:20分，选择题每小题4分）}

\ansitem{1}{C}
\ansline{解析}{电动势 $1.5\,\unit{V}$ 的物理意义是非静电力每把 $1\,\unit{C}$ 正电荷从负极搬运到正极做功 $1.5\,\unit{J}$，即把 $1.5\,\unit{J}$ 化学能转化为电能，C正确；电池储存的化学能量与电动势无直接对应关系，A错误；接入电路后路端电压 $U<E$，B错误；内阻也消耗电能，外电路每通过 $1\,\unit{C}$ 消耗的电能小于 $1.5\,\unit{J}$，D错误．}

\ansitem{2}{A}
\ansline{解析}{$I=\frac{E}{R+r}=\frac{3}{5.5+0.5}\,\unit{A}=0.5\,\unit{A}$，$U=IR=0.5\times5.5\,\unit{V}=2.75\,\unit{V}$（$U=E-Ir=3-0.25\,\unit{V}=2.75\,\unit{V}$），A正确，B、C、D错误．}

\ansitem{3}{C}
\ansline{解析}{断路时 $I=0$，路端电压 $U=E$，A错误；短路时路端电压为零，B错误；由 $U=E-Ir$ 知 $R$ 增大时 $I$ 减小、$U$ 增大，且恒有 $U<E$，C正确；电源的效率 $\eta=\frac{U}{E}$ 随 $R$ 增大而增大，D错误．}

\ansitem{4}{$1\,\unit{A}$；$5\,\unit{V}$}
\ansline{解析}{$I=\frac{E}{R+r}=\frac{6}{5+1}\,\unit{A}=1\,\unit{A}$，$U=E-Ir=6-1\times1\,\unit{V}=5\,\unit{V}$（或 $U=IR=1\times5\,\unit{V}=5\,\unit{V}$）．}

\ansitem{5}{$8\,\unit{V}$；$16\,\unit{W}$}
\ansline{解析}{$I=\frac{E}{R+r}=\frac{10}{4+1}\,\unit{A}=2\,\unit{A}$，$U=E-Ir=10-2\times1\,\unit{V}=8\,\unit{V}$，$P_{\text{出}}=UI=8\times2\,\unit{W}=16\,\unit{W}$．}
